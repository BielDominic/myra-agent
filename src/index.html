// ============================================================ // MYRA Widget — Script (versão Tauri) // ============================================================ const AGENT_URL = "http://127.0.0.1:8000"; // Dimensões const IDLE_WIDTH = 200; const IDLE_HEIGHT = 44; const CARD_WIDTH = 420; const CARD_HEIGHT = 260; let lastState = null; let startTime = null; // ============================================================ // IPC helper // ============================================================ async function invoke(cmd, args = {}) { if (window.__TAURI__ && window.__TAURI__.core) { return await window.__TAURI__.core.invoke(cmd, args); } console.warn("[TAURI] API não disponível. Rodando em modo browser?"); return null; } // ============================================================ // Resize da janela via Tauri // ============================================================ async function setWindowSize(state) { let w, h; if (state === "idle") { w = IDLE_WIDTH; h = IDLE_HEIGHT; } else { w = CARD_WIDTH; h = CARD_HEIGHT; } try { await invoke("resize_window", { width: w, height: h }); console.log(`[RESIZE] ${state} -> ${w}x${h}`); } catch (e) { console.error("[RESIZE] erro:", e); } } // ============================================================ // Troca de view // ============================================================ function switchView(state) { const island = document.getElementById("island"); if (!island) return; // Remove classes antigas island.classList.remove("idle", "running", "done", "error"); island.classList.add(state); // Troca views ativas document.querySelectorAll(".view").forEach((v) => v.classList.remove("active")); const view = document.querySelector(`.view-${state}`); if (view) view.classList.add("active"); // Resize da janela setWindowSize(state); } // ============================================================ // Ações // ============================================================ function closeWidget() { console.log("[WIDGET] close"); switchView("idle"); lastState = "idle"; } function resetWidget() { console.log("[WIDGET] reset"); switchView("idle"); lastState = "idle"; } async function cancelTask() { console.log("[WIDGET] cancel"); try { await invoke("cancel_task"); } catch (e) { console.error("[CANCEL] erro:", e); } } // ============================================================ // Polling de status // ============================================================ async function pollStatus() { try { const status = await invoke("get_task_status"); if (!status) return; const state = status.state || "idle"; if (state !== lastState) { console.log(`[POLL] state: ${lastState} -> ${state}`); lastState = state; // Atualiza campos if (state === "running") { const tool = document.getElementById("tool-name"); if (tool && status.passo_atual) tool.textContent = status.passo_atual; const tempo = document.getElementById("meta-tempo"); if (tempo && status.elapsed !== undefined) { tempo.textContent = `⏱ ${Math.floor(status.elapsed)}s`; } const passos = document.getElementById("meta-passos"); if (passos && status.passo_num !== undefined && status.passo_total !== undefined) { passos.textContent = `⚙ ${status.passo_num}/${status.passo_total}`; } } if (state === "done") { const result = document.getElementById("result-text"); if (result) result.textContent = status.resultado || "(sem resultado)"; } if (state === "error") { const err = document.getElementById("error-text"); if (err) err.textContent = status.erro || "(sem mensagem)"; } switchView(state); } else if (state === "running") { // Atualiza campos mesmo se o estado não mudou (tempo, passo) const tool = document.getElementById("tool-name"); if (tool && status.passo_atual) tool.textContent = status.passo_atual; const tempo = document.getElementById("meta-tempo"); if (tempo && status.elapsed !== undefined) { tempo.textContent = `⏱ ${Math.floor(status.elapsed)}s`; } const passos = document.getElementById("meta-passos"); if (passos && status.passo_num !== undefined && status.passo_total !== undefined) { passos.textContent = `⚙ ${status.passo_num}/${status.passo_total}`; } } } catch (e) { // Agent offline — silêncio } } // ============================================================ // Init // ============================================================ window.addEventListener("DOMContentLoaded", () => { console.log("[WIDGET] init"); switchView("idle"); lastState = "idle"; // Polling a cada 500ms setInterval(pollStatus, 500); });
